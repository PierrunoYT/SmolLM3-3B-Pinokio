# SmolLM3-3B Chatbot Requirements
# Note: PyTorch installation is handled by torch.js based on platform/GPU,
# so torch is intentionally not pinned here.

# Core ML dependencies
# torch.js installs torch 2.2.2 on Intel Macs and torch 2.4.1 (torch-directml)
# on AMD Windows. Transformers 5.x silently disables PyTorch below its floor
# (2.6 as of 5.20), so stay on 4.x, which supports torch>=2.2.
transformers>=4.56.0,<5
accelerate
tokenizers
safetensors

# Web interface
gradio>=4.0.0

# Text processing and model support
sentencepiece
protobuf
huggingface-hub

# Utility libraries
# Intel Macs get torch 2.2.2 from torch.js, which was built against NumPy 1.x.
numpy<2; sys_platform == "darwin" and platform_machine == "x86_64"
numpy; sys_platform != "darwin" or platform_machine != "x86_64"
packaging
pyyaml
regex
requests
tqdm
